    <script>
        let semanaOffset = 0; // 0 = actual, 1 = siguiente
        let fechaActivaStr = '';

        const NOMBRES_DIAS = ['Lunes', 'Martes', 'Miércoles', 'Jueves', 'Viernes', 'Sábado', 'Domingo'];

        // Obtener el Lunes de la semana indicada según offset
        function calcularLunes(offset) {
            const fecha = new Date();
            const diaSemana = fecha.getDay(); // 0 es Domingo
            const diferencia = (diaSemana === 0 ? -6 : 1) - diaSemana;
            fecha.setDate(fecha.getDate() + diferencia + (offset * 7));
            return fecha;
        }

        // Construir la barra de días bloqueando días anteriores automáticamente
        function construirBarraDias() {
            const barra = document.getElementById('barraDias');
            barra.innerHTML = '';

            const lunes = calcularLunes(semanaOffset);
            const hoyStr = new Date().toLocaleDateString('en-CA'); // 'YYYY-MM-DD'
            let primerDiaHabilitado = null;

            for (let i = 0; i < 7; i++) {
                const fechaDia = new Date(lunes);
                fechaDia.setDate(lunes.getDate() + i);
                const str = fechaDia.toLocaleDateString('en-CA');

                const esPasado = str < hoyStr;
                const esHoy = str === hoyStr;

                if (!esPasado && !primerDiaHabilitado) {
                    primerDiaHabilitado = str;
                }

                const card = document.createElement('div');
                card.className = `dia-card ${esPasado ? 'bloqueado' : ''} ${str === fechaActivaStr ? 'activo' : ''}`;

                let badgeHoyHtml = esHoy ? `<span class="badge-hoy">HOY</span>` : '';

                card.innerHTML = `
                    ${badgeHoyHtml}
                    <span class="etiqueta-nombre">${NOMBRES_DIAS[i]}</span>
                    <span class="numero-fecha">${fechaDia.getDate()}</span>
                `;

                if (!esPasado) {
                    card.onclick = () => fijarDiaActivo(str);
                }

                barra.appendChild(card);
            }

            // Seleccionar por defecto hoy o el primer día habilitado
            if (!fechaActivaStr || fechaActivaStr < hoyStr || semanaOffset === 1 && !fechaActivaStr) {
                fijarDiaActivo(semanaOffset === 0 ? hoyStr : primerDiaHabilitado);
            }
        }

        function seleccionarSemana(offset) {
            semanaOffset = offset;
            document.getElementById('btnSemanaActual').classList.toggle('activo', offset === 0);
            document.getElementById('btnSemanaSiguiente').classList.toggle('activo', offset === 1);
            fechaActivaStr = '';
            construirBarraDias();
        }

        function fijarDiaActivo(str) {
            fechaActivaStr = str;
            const tarjetas = document.querySelectorAll('.dia-card');
            const lunes = calcularLunes(semanaOffset);

            tarjetas.forEach((tarjeta, index) => {
                const dia = new Date(lunes);
                dia.setDate(lunes.getDate() + index);
                const currentStr = dia.toLocaleDateString('en-CA');
                tarjeta.classList.toggle('activo', currentStr === fechaActivaStr);
            });

            cargarGrillaTurnos();
        }

        // Cargar los 24 bloques horarios con detección de horas pasadas
        async function cargarGrillaTurnos() {
            if (!fechaActivaStr) return;
            const contenedor = document.getElementById('grillaTurnos');
            contenedor.innerHTML = '<p style="color: var(--texto-apagado); grid-column: 1/-1;">Cargando horarios...</p>';

            try {
                const res = await fetch(`/api/horarios?fecha=${fechaActivaStr}`);
                const data = await res.json();
                if (!res.ok) throw new Error(data.error);

                contenedor.innerHTML = '';

                const hoyStr = new Date().toLocaleDateString('en-CA');
                const horaActual = new Date().getHours();
                const esHoy = fechaActivaStr === hoyStr;

                data.bloques.forEach(b => {
                    const card = document.createElement('div');
                    
                    // Bloquear horas pasadas del día actual automáticamente
                    const bloquePasado = esHoy && b.hora < horaActual;

                    let claseEstado = '';
                    if (bloquePasado) claseEstado = 'pasado';
                    else if (b.reservadoPorMi) claseEstado = 'mio';
                    else if (!b.disponible) claseEstado = 'lleno';

                    card.className = `turno-card ${claseEstado}`;

                    // Participantes para administradores (Nombre, Cargo y Área separados)
                    let participantesHtml = '';
                    if (b.participantes && b.participantes.length > 0) {
                        participantesHtml = `
                            <div class="participantes-caja">
                                ${b.participantes.map(p => `<div>• ${p.nombre} (${p.cargo} | ${p.area})</div>`).join('')}
                            </div>
                        `;
                    }

                    // Determinar botón de acción
                    let botonAccion = '';
                    if (bloquePasado) {
                        botonAccion = `<button class="btn-turno btn-deshabilitado" disabled>Horario Pasado</button>`;
                    } else if (b.reservadoPorMi) {
                        botonAccion = `<button class="btn-turno btn-cancelar" onclick="cancelarTurno(${b.reservaId})">Cancelar Mi Reserva</button>`;
                    } else if (b.disponible) {
                        botonAccion = `<button class="btn-turno btn-reservar" onclick="reservarTurno(${b.hora})">Reservar Cupo</button>`;
                    } else {
                        botonAccion = `<button class="btn-turno btn-deshabilitado" disabled>Aforo Completo (4/4)</button>`;
                    }

                    // Badge de aforo
                    let badgeClase = 'disponible';
                    if (bloquePasado) badgeClase = 'pasado';
                    else if (b.ocupados >= b.capacidadMaxima) badgeClase = 'lleno';

                    card.innerHTML = `
                        <div>
                            <div class="turno-encabezado">
                                <span class="turno-hora">${b.horaTexto}</span>
                                <span class="badge-aforo ${badgeClase}">
                                    ${bloquePasado ? 'Finalizado' : `${b.ocupados} / ${b.capacidadMaxima}`}
                                </span>
                            </div>
                            ${participantesHtml}
                        </div>
                        ${botonAccion}
                    `;

                    contenedor.appendChild(card);
                });

            } catch (err) {
                contenedor.innerHTML = `<p style="color: var(--rojo-alerta); grid-column: 1/-1;">Error al sincronizar: ${err.message}</p>`;
            }
        }

        // Ejecutar Reserva
        async function reservarTurno(hora) {
            try {
                const res = await fetch('/api/reservar', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ fecha: fechaActivaStr, horaInicio: hora })
                });

                const data = await res.json();
                if (!res.ok) throw new Error(data.error);

                mostrarToast("✓ " + data.mensaje);
                cargarGrillaTurnos();
            } catch (err) {
                alert("No se pudo reservar: " + err.message);
            }
        }

        // Cancelar Reserva
        async function cancelarTurno(reservaId) {
            if (!confirm("¿Deseas cancelar tu reserva de este bloque horario?")) return;

            try {
                const res = await fetch('/api/cancelar-reserva', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ reservaId })
                });

                const data = await res.json();
                if (!res.ok) throw new Error(data.error);

                mostrarToast("✓ Reserva cancelada.");
                cargarGrillaTurnos();
            } catch (err) {
                alert("Error al cancelar: " + err.message);
            }
        }

        // Notificaciones SSE en Tiempo Real (Cancelaciones de Hoy o Mañana)
        if (!!window.EventSource) {
            const canalSSE = new EventSource('/api/notificaciones/stream');

            canalSSE.onmessage = function(event) {
                try {
                    const data = JSON.parse(event.data);
                    const mensaje = `Alguien ha cancelado una hora.\nDía: ${data.dia}\nHora: ${data.hora}`;

                    mostrarToast(mensaje);

                    if (window.Notification && Notification.permission === "granted") {
                        new Notification("Hora Cancelada en Gimnasio", {
                            body: `Se ha cancelado una hora para ${data.dia} a las ${data.hora}.`,
                            icon: "/img/logo.png"
                        });
                    }

                    cargarGrillaTurnos();
                } catch (e) {
                    console.error("Error SSE:", e);
                }
            };
        }

        function solicitarPermisoAvisos() {
            if (!("Notification" in window)) {
                alert("Navegador sin soporte de notificaciones.");
                return;
            }
            Notification.requestPermission().then(p => {
                if (p === "granted") {
                    document.getElementById('btnNotif').style.display = 'none';
                    mostrarToast("Avisos activados con éxito.");
                }
            });
        }

        if (window.Notification && Notification.permission === "granted") {
            document.getElementById('btnNotif').style.display = 'none';
        }

        function mostrarToast(txt) {
            const toast = document.createElement('div');
            toast.className = 'alerta-toast';
            toast.innerText = txt;
            document.body.appendChild(toast);

            setTimeout(() => {
                toast.style.opacity = '0';
                toast.style.transform = 'translateY(10px)';
                setTimeout(() => toast.remove(), 400);
            }, 5000);
        }

        // Inicio automático en Semana Actual
        construirBarraDias();
    </script>
